function runValidation() {

  const results = document.getElementById("validatorResults");

  const msgs = [];

  const summaryLines = [];


  const addMsg = (type, text, rawForAI) => {

    msgs.push(`<div class="alert ${type}">${text}</div>`);

    summaryLines.push(`[${type.toUpperCase()}] ${rawForAI}`);

  };


  if (currentPracticeId === "prac_pesos") {

    const n = parseFloat(document.getElementById("v_n")?.value);

    const rMean = parseFloat(document.getElementById("v_rook_mean")?.value);

    const rPct = parseFloat(document.getElementById("v_rook_pct")?.value);

    const qMean = parseFloat(document.getElementById("v_queen_mean")?.value);

    const qPct = parseFloat(document.getElementById("v_queen_pct")?.value);

    const q2Pure = parseFloat(document.getElementById("v_q2_pure")?.value);

    const q2Inc = parseFloat(document.getElementById("v_q2_inc")?.value);


    if (!isNaN(n) && n > 0 && !isNaN(rMean) && !isNaN(rPct)) {

      const expected = (rMean / n) * 100;

      if (Math.abs(expected - rPct) <= 0.15) {

        addMsg("ok", `✅ <strong>Rook coherente:</strong> Con n=${n} y media=${rMean}, el % no cero esperado es ~${expected.toFixed(2)}% (reportaste ${rPct}%).`, `Rook % no cero coherente (${rPct}% vs ${expected.toFixed(2)}%).`);

      } else {

        addMsg("err", `⚠️ <strong>Revisa tu % no cero de Rook:</strong> Con n=${n} y media=${rMean}, la relación (media/n)×100 da <strong>${expected.toFixed(2)}%</strong>, pero ingresaste ${rPct}%. Revisa si copiaste mal el dato o n.`, `Inconsistencia en Rook % no cero: reportó ${rPct}%, pero (media=${rMean}/n=${n})*100 = ${expected.toFixed(2)}%.`);

      }

    }


    if (!isNaN(n) && n > 0 && !isNaN(qMean) && !isNaN(qPct)) {

      const expected = (qMean / n) * 100;

      if (Math.abs(expected - qPct) <= 0.15) {

        addMsg("ok", `✅ <strong>Queen coherente:</strong> Con n=${n} y media=${qMean}, el % no cero esperado es ~${expected.toFixed(2)}% (reportaste ${qPct}%).`, `Queen % no cero coherente (${qPct}%).`);

      } else {

        addMsg("err", `⚠️ <strong>Revisa tu % no cero de Queen:</strong> Con n=${n} y media=${qMean}, (media/n)×100 da <strong>${expected.toFixed(2)}%</strong>, pero ingresaste ${qPct}%.`, `Inconsistencia en Queen % no cero: reportó ${qPct}%, esperado ${expected.toFixed(2)}%.`);

      }

    }


    if (!isNaN(rMean) && !isNaN(qMean)) {

      if (qMean >= rMean) {

        addMsg("ok", `✅ <strong>Relación Queen ≥ Rook correcta:</strong> Queen (${qMean}) tiene al menos tantos vecinos en promedio como Rook (${rMean}).`, `Relación Queen >= Rook válida (${qMean} >= ${rMean}).`);

      } else {

        addMsg("err", `⚠️ <strong>Error conceptual/captura:</strong> Queen (${qMean}) no puede tener menor media de vecinos que Rook (${rMean}), porque Queen incluye bordes y vértices.`, `Error: reportó media Queen (${qMean}) < media Rook (${rMean}).`);

      }

    }


    if (!isNaN(qMean) && !isNaN(q2Pure) && !isNaN(q2Inc)) {

      const sum = qMean + q2Pure;

      if (Math.abs(sum - q2Inc) <= 0.25) {

        addMsg("ok", `✅ <strong>Orden 2 inclusivo coherente:</strong> Orden 1 (${qMean}) + Orden 2 puro (${q2Pure}) ≈ ${sum.toFixed(2)}, consistente con la inclusiva (${q2Inc}).`, `Orden 2 inclusivo coherente (${q2Inc} ≈ ${qMean} + ${q2Pure}).`);

      } else {

        addMsg("err", `⚠️ <strong>Revisa Orden 2 inclusivo:</strong> La media inclusiva debería ser aproximadamente la suma de Orden 1 (${qMean}) + Orden 2 sin inferiores (${q2Pure}) ≈ <strong>${sum.toFixed(2)}</strong>, pero reportaste ${q2Inc}.`, `Inconsistencia en Orden 2 inclusivo: reportó ${q2Inc}, pero Orden 1 + Orden 2 puro = ${sum.toFixed(2)}.`);

      }

    }

  } else if (currentPracticeId === "prac2_baltimore") {

    const bn = parseFloat(document.getElementById("v_b_n")?.value);

    const bf = parseFloat(document.getElementById("v_b_filter")?.value);

    const bp = parseFloat(document.getElementById("v_b_ppsqft")?.value);


    if (!isNaN(bn)) {

      if (bn === 211) addMsg("ok", `✅ <strong>Total de observaciones correcto:</strong> 211 propiedades.`, `n=211 correcto.`);

      else addMsg("err", `⚠️ <strong>Revisa el total de observaciones:</strong> La base original baltim tiene 211 propiedades (reportaste ${bn}).`, `Reportó n=${bn} en vez de 211.`);

    }

    if (!isNaN(bf)) {

      if (bf === 61) {

        addMsg("ok", `✅ <strong>Filtro correcto:</strong> 61 propiedades tienen entre 2 y 3 baños (incluyendo 2, 2.5 y 3).`, `Filtro de baños correcto (61).`);

      } else if (bf === 41) {

        addMsg("err", `⚠️ <strong>¡Cuidado con los medios baños!</strong> Obtuviste 41 propiedades porque probablemente convertiste <code>NBATH</code> a entero (integer) y perdiste las casas con <strong>2.5 baños</strong>. Revisa el paso 2.`, `Error clásico en filtro de baños: obtuvo 41 en lugar de 61 por convertir NBATH a entero y perder los valores 2.5.`);

      } else {

        addMsg("warn", `⚠️ <strong>Verifica tu filtro:</strong> Reportaste ${bf} propiedades. Asegúrate de incluir los extremos (<code>NBATH >= 2</code> y <code>NBATH <= 3</code>) y de no haber truncado los medios baños (2.5).`, `Reportó ${bf} propiedades en el filtro de baños (esperado: 61).`);

      }

    }

    if (!isNaN(bp)) {

      if (bp >= 2.5 && bp <= 3.5) {

        addMsg("ok", `✅ <strong>Orden de magnitud de PRICE/SQFT correcto (~${bp}):</strong> Recuerda explicar en tu texto en qué unidades están PRICE y SQFT (no son dólares directos por pie cuadrado).`, `Media PRICE/SQFT razonable (${bp}).`);

      } else if (bp > 0 && bp < 1) {

        addMsg("err", `⚠️ <strong>Posible división invertida:</strong> Tu media (${bp}) sugiere que dividiste <code>SQFT / PRICE</code> en lugar de <code>PRICE / SQFT</code>.`, `Posible división invertida en PRICE/SQFT (media=${bp}).`);

      }

    }

  } else if (currentPracticeId === "prac5_mapas") {

    const b15 = parseFloat(document.getElementById("v_m_box15")?.value);

    const b30 = parseFloat(document.getElementById("v_m_box30")?.value);

    const eq1 = parseFloat(document.getElementById("v_m_eq1")?.value);


    if (!isNaN(b15) && !isNaN(b30)) {

      if (b30 <= b15) {

        addMsg("ok", `✅ <strong>Relación lógica de Box Map correcta:</strong> Con hinge 3.0 (${b30} outliers) hay igual o menos atípicos que con hinge 1.5 (${b15}).`, `Relación Box Map válida: hinge 3.0 (${b30}) <= hinge 1.5 (${b15}).`);

      } else {

        addMsg("err", `⚠️ <strong>Inconsistencia en Box Map:</strong> Con hinge 3.0 las cercas son más amplias, así que nunca puede haber más outliers (${b30}) que con hinge 1.5 (${b15}).`, `Error lógico en Box Map: reportó más outliers en hinge 3.0 (${b30}) que en 1.5 (${b15}).`);

      }

    }

    if (!isNaN(eq1)) {

      if (eq1 >= 70 && eq1 <= 78) {

        addMsg("ok", `✅ <strong>Conteo coherente en Equal Intervals (${eq1}/78):</strong> Nota cómo casi todos los condados quedan agrupados en la primera categoría debido al sesgo.`, `Equal intervals cat 1 = ${eq1}/78.`);

    }

  results.innerHTML = msgs.join("");

  lastValidationSummary = summaryLines.join("\n");

document.getElementById("fileInput").addEventListener("change", async function(e) {

  const file = e.target.files[0];

  if (!file) return;


  const ext = file.name.split(".").pop().toLowerCase();

  const draftBox = document.getElementById("studentDraft");

  const imgWarn = document.getElementById("imageWarning");

  imgWarn.classList.add("hidden");


  try {

    if (ext === "txt" || ext === "md") {

      const text = await file.text();

      draftBox.value = text;

    } else if (ext === "docx") {

      imgWarn.classList.remove("hidden");

      const arrayBuffer = await file.arrayBuffer();

      const result = await mammoth.extractRawText({ arrayBuffer });

      draftBox.value = result.value;

    } else if (ext === "pdf") {

      imgWarn.classList.remove("hidden");

      const arrayBuffer = await file.arrayBuffer();

      const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;

      let fullText = "";

      for (let i = 1; i <= pdf.numPages; i++) {

        const page = await pdf.getPage(i);

        const content = await page.getTextContent();

        fullText += content.items.map(item => item.str).join(" ") + "\n\n";

      }

      draftBox.value = fullText.trim();

    } else {

      alert("Formato no soportado. Usa .docx, .pdf, .txt o .md.");

    }

    updateCharCount();

  } catch (err) {

    console.error(err);

    alert("Hubo un problema al leer el archivo. Intenta copiar y pegar el texto directamente en el cuadro.");

  }

});
